use wayroll;
SET @targetCurrency = 'USD';
SET @childCustomer = 0;
SET @isCrossDivision = FALSE;
SET @revenueModel = 'wayroll-gross-mrr';
SET @currentDate = '2026-05-15';
SET @curMonth = '2026-05-01';
SET @isLeapYear = FALSE;
SET @projectId = 1000054;
SET @hashStart = NULL;
SET @hashEnd = NULL;
SET @modelType = 'mrr';
SET @includeEstimates = 1;
SET @comparisonMonths = 1;

WITH invoice_base AS (
  SELECT
    projectId,
    divisionId,
    customerNumberHash,
    date,
    originalAmount,
    invoiceId,
    invoiceDate,
    startDate,
    originalCustomerNumber,
    originalCustomerName,
    originalCurrency,
    planId,
    lineItemType,
    integrationConfigId,
    NULL AS parentCustomerNumber,
    NULL AS parentCustomerName,
    customerNumber AS customerNumber,
    LOWER(customerNumber) AS customerNumberLC,
    customerName AS customerName,
    customerEmail AS customerEmail,
    amount
  FROM invoice_import_amortized
  WHERE projectId = @projectId AND modelType = @modelType AND deleted = 0 AND date <= DATE_ADD(CURDATE(), INTERVAL 3 YEAR) AND lineItemType IN ('credit-prorated','discount-recurring','revenue-recurring','adjustment-recurring','revenue-nonrecurring','adjustment-recurring','adjustment-nonrecurring') 
  AND divisionId IN (1000093)
)
, last_month_invoices AS (
  SELECT
    projectId,
    divisionId,
    customerNumberHash,
    date,
    originalAmount,
    invoiceId,
    invoiceDate,
    startDate,
    originalCustomerNumber,
    originalCustomerName,
    originalCurrency,
    planId,
    lineItemType,
    integrationConfigId,
    parentCustomerNumber,
    parentCustomerName,
    customerNumber,
    customerNumberLC,
    customerName,
    customerEmail,
    amount
  FROM invoice_base
  WHERE date BETWEEN DATE_ADD(@curMonth, INTERVAL -1 MONTH) AND DATE_ADD(@curMonth, INTERVAL -1 DAY)
)
, customer_3month_aggregates AS (
  SELECT
    projectId,
    divisionId,
    customerNumberLC,
    COUNT(DISTINCT CASE WHEN date BETWEEN @curMonth AND LAST_DAY(@curMonth) THEN invoiceId END) AS thisMonthInvoiceCount,
    COUNT(DISTINCT CASE WHEN date BETWEEN DATE_ADD(@curMonth, INTERVAL -1 MONTH) AND DATE_ADD(@curMonth, INTERVAL -1 DAY) THEN invoiceId END) AS lastMonthInvoiceCount,
    COUNT(DISTINCT CASE WHEN date BETWEEN DATE_ADD(@curMonth, INTERVAL -2 MONTH) AND DATE_ADD(DATE_ADD(@curMonth, INTERVAL -1 MONTH), INTERVAL -1 DAY) THEN invoiceId END) AS twoMonthsAgoInvoiceCount,
    MAX_BY(startDate, CASE WHEN date BETWEEN DATE_ADD(@curMonth, INTERVAL -2 MONTH) AND DATE_ADD(@curMonth, INTERVAL -1 DAY) THEN DAY(startDate) ELSE 0 END) AS maxStartDate,
    MAX_BY(invoiceDate, CASE WHEN date BETWEEN DATE_ADD(@curMonth, INTERVAL -2 MONTH) AND DATE_ADD(@curMonth, INTERVAL -1 DAY) THEN DAY(invoiceDate) ELSE 0 END) AS maxInvoiceDate
  FROM invoice_base
  WHERE date BETWEEN DATE_ADD(@curMonth, INTERVAL -2 MONTH) AND LAST_DAY(@curMonth) AND lineItemType <> 'revenue-nonrecurring'
  GROUP BY projectId, divisionId, customerNumberLC
)
, last_month_invoices_with_adjusted_dates AS (
  SELECT
    last_month_invoices.projectId,
    last_month_invoices.divisionId,
    last_month_invoices.customerNumberHash,
    last_month_invoices.date,
    last_month_invoices.originalAmount,
    last_month_invoices.invoiceId,
    last_month_invoices.invoiceDate,
    last_month_invoices.startDate,
    last_month_invoices.originalCustomerNumber,
    last_month_invoices.originalCustomerName,
    last_month_invoices.originalCurrency,
    last_month_invoices.planId,
    last_month_invoices.lineItemType,
    last_month_invoices.integrationConfigId,
    last_month_invoices.parentCustomerNumber,
    last_month_invoices.parentCustomerName,
    last_month_invoices.customerNumber,
    last_month_invoices.customerNumberLC,
    last_month_invoices.customerName,
    last_month_invoices.customerEmail,
    last_month_invoices.amount,
    customer_3month_aggregates.lastMonthInvoiceCount,
    customer_3month_aggregates.thisMonthInvoiceCount,
    customer_3month_aggregates.twoMonthsAgoInvoiceCount,
    CASE 
              WHEN customer_3month_aggregates.lastMonthInvoiceCount = 1 AND customer_3month_aggregates.twoMonthsAgoInvoiceCount = 1 AND DAY(last_month_invoices.startDate) >= 28 THEN COALESCE(customer_3month_aggregates.maxStartDate, last_month_invoices.startDate) 
              ELSE last_month_invoices.startDate END AS finalStartDate,
    CASE 
              WHEN customer_3month_aggregates.lastMonthInvoiceCount = 1 AND customer_3month_aggregates.twoMonthsAgoInvoiceCount = 1 AND DAY(last_month_invoices.invoiceDate) >= 28 THEN COALESCE(customer_3month_aggregates.maxInvoiceDate, last_month_invoices.invoiceDate) 
              ELSE last_month_invoices.invoiceDate END AS finalInvoiceDate
  FROM last_month_invoices
  LEFT JOIN customer_3month_aggregates ON customer_3month_aggregates.projectId = last_month_invoices.projectId 
           AND customer_3month_aggregates.divisionId = last_month_invoices.divisionId AND customer_3month_aggregates.customerNumberLC = last_month_invoices.customerNumberLC
)
, estimate_candidates AS (
  SELECT
    last_month_invoices_with_adjusted_dates.projectId,
    last_month_invoices_with_adjusted_dates.divisionId,
    last_month_invoices_with_adjusted_dates.customerNumberHash,
    last_month_invoices_with_adjusted_dates.date,
    last_month_invoices_with_adjusted_dates.originalAmount,
    last_month_invoices_with_adjusted_dates.invoiceId,
    last_month_invoices_with_adjusted_dates.invoiceDate,
    last_month_invoices_with_adjusted_dates.startDate,
    last_month_invoices_with_adjusted_dates.originalCustomerNumber,
    last_month_invoices_with_adjusted_dates.originalCustomerName,
    last_month_invoices_with_adjusted_dates.originalCurrency,
    last_month_invoices_with_adjusted_dates.planId,
    last_month_invoices_with_adjusted_dates.lineItemType,
    last_month_invoices_with_adjusted_dates.integrationConfigId,
    last_month_invoices_with_adjusted_dates.parentCustomerNumber,
    last_month_invoices_with_adjusted_dates.parentCustomerName,
    last_month_invoices_with_adjusted_dates.customerNumber,
    last_month_invoices_with_adjusted_dates.customerNumberLC,
    last_month_invoices_with_adjusted_dates.customerName,
    last_month_invoices_with_adjusted_dates.customerEmail,
    last_month_invoices_with_adjusted_dates.amount,
    last_month_invoices_with_adjusted_dates.lastMonthInvoiceCount,
    last_month_invoices_with_adjusted_dates.thisMonthInvoiceCount,
    last_month_invoices_with_adjusted_dates.twoMonthsAgoInvoiceCount,
    last_month_invoices_with_adjusted_dates.finalStartDate,
    last_month_invoices_with_adjusted_dates.finalInvoiceDate,
    IF(bd.targetDay IS NULL, 
          DATE_ADD(DATE_ADD(MAKEDATE(YEAR(@curMonth), 1), INTERVAL (MONTH(@curMonth) - 1) MONTH), INTERVAL (DAY(last_month_invoices_with_adjusted_dates.finalStartDate) - 1) DAY), 
          DATE_ADD(DATE_ADD(MAKEDATE(YEAR(@curMonth), 1), INTERVAL (MONTH(@curMonth) - 1) MONTH), INTERVAL (bd.targetDay - 1) DAY)
        ) AS estimatedBillDate,
    IF(id.targetDay IS NULL, 
          DATE_ADD(DATE_ADD(MAKEDATE(YEAR(@curMonth), 1), INTERVAL (MONTH(@curMonth) - 1) MONTH), INTERVAL (DAY(last_month_invoices_with_adjusted_dates.finalInvoiceDate) - 1) DAY), 
          DATE_ADD(DATE_ADD(MAKEDATE(YEAR(@curMonth), 1), INTERVAL (MONTH(@curMonth) - 1) MONTH), INTERVAL (id.targetDay - 1) DAY)
        ) AS estimatedInvoiceDate
  FROM last_month_invoices_with_adjusted_dates
  LEFT JOIN estimate_date_map bd ON MONTH(last_month_invoices_with_adjusted_dates.finalStartDate) = bd.sourceMonth AND DAY(last_month_invoices_with_adjusted_dates.finalStartDate) = bd.sourceDay AND bd.targetMonth = MONTH(@curMonth) AND bd.leapYear = @isLeapYear 
           LEFT JOIN estimate_date_map id ON MONTH(last_month_invoices_with_adjusted_dates.finalInvoiceDate) = id.sourceMonth AND DAY(last_month_invoices_with_adjusted_dates.finalInvoiceDate) = id.sourceDay AND id.targetMonth = MONTH(@curMonth) AND id.leapYear = @isLeapYear
  WHERE 1 = @includeEstimates
)
, invoice_estimate_sequence AS (
  SELECT
    estimate_candidates.projectId,
    estimate_candidates.divisionId,
    estimate_candidates.customerNumberLC,
    estimate_candidates.invoiceId,
    ANY_VALUE(estimate_candidates.estimatedInvoiceDate) AS estimatedInvoiceDate,
    ROW_NUMBER() OVER (PARTITION BY estimate_candidates.projectId, estimate_candidates.divisionId, 
                 estimate_candidates.customerNumberLC ORDER BY ANY_VALUE(estimate_candidates.estimatedInvoiceDate) DESC, 
                 estimate_candidates.invoiceId) AS invoice_sequence
  FROM estimate_candidates
  GROUP BY estimate_candidates.projectId, estimate_candidates.divisionId, estimate_candidates.customerNumberLC, estimate_candidates.invoiceId
)

    estimate_candidates.customerEmail,
    estimate_candidates.amount,

    estimate_candidates.finalStartDate,
    estimate_candidates.finalInvoiceDate,
    estimate_candidates.estimatedBillDate,
    estimate_candidates.estimatedInvoiceDate,
    invoice_estimate_sequence.invoice_sequence
  FROM estimate_candidates
  INNER JOIN invoice_estimate_sequence ON invoice_estimate_sequence.projectId = estimate_candidates.projectId 
           AND invoice_estimate_sequence.divisionId = estimate_candidates.divisionId AND invoice_estimate_sequence.customerNumberLC = estimate_candidates.customerNumberLC AND invoice_estimate_sequence.invoiceId = estimate_candidates.invoiceId
  WHERE (estimate_candidates.estimatedBillDate >= @currentDate OR estimate_candidates.estimatedInvoiceDate >= @currentDate) AND (thisMonthInvoiceCount + invoice_sequence) <= lastMonthInvoiceCount
)
, estimates_transformed AS (
  SELECT
    estimates_final.projectId,
    estimates_final.divisionId,
    estimates_final.customerNumberHash,
    LAST_DAY(estimates_final.estimatedBillDate) AS date,
    estimates_final.originalAmount,
    CONCAT(estimates_final.invoiceId, '-estimate') AS invoiceId,
    estimates_final.estimatedInvoiceDate AS invoiceDate,
    estimates_final.estimatedBillDate AS startDate,
    estimates_final.originalCustomerNumber,
    estimates_final.originalCustomerName,
    estimates_final.originalCurrency,
    estimates_final.planId,
    estimates_final.lineItemType,
    estimates_final.integrationConfigId,
    estimates_final.parentCustomerNumber,
    estimates_final.parentCustomerName,
    estimates_final.customerNumber,
    estimates_final.customerNumberLC,
    estimates_final.customerName,
    estimates_final.customerEmail,
    estimates_final.amount
  FROM estimates_final
)

)
, converted_invoice_normalized_handler AS (
  SELECT
    with_normalized_currency.projectId,
    with_normalized_currency.divisionId,
    with_normalized_currency.customerNumberHash,
    with_normalized_currency.date,
    with_normalized_currency.originalAmount,
    with_normalized_currency.invoiceId,
    with_normalized_currency.invoiceDate,
    with_normalized_currency.startDate,
    with_normalized_currency.originalCustomerNumber,
    with_normalized_currency.originalCustomerName,
    with_normalized_currency.originalCurrency,
    with_normalized_currency.planId,
    with_normalized_currency.lineItemType,
    with_normalized_currency.integrationConfigId,
    with_normalized_currency.parentCustomerNumber,
    with_normalized_currency.parentCustomerName,
    with_normalized_currency.customerNumber,
    with_normalized_currency.customerNumberLC,
    with_normalized_currency.customerName,
    with_normalized_currency.customerEmail,
    with_normalized_currency.amount,
    with_normalized_currency.normalizedCurrency,
    CAST(
              CASE
                 WHEN with_normalized_currency.normalizedCurrency = 'USD' THEN with_normalized_currency.amount
                 ELSE CAST(ROUND((with_normalized_currency.originalAmount * (CAST(COALESCE(c_out.rate, 1) / COALESCE(c_in.rate, 1) AS DOUBLE)))) AS INT)
              END AS INT) AS originalAmountNormalized
  FROM with_normalized_currency
  LEFT JOIN currency_exchange_rate c_in ON c_in.currencyTo = with_normalized_currency.originalCurrency AND c_in.currencyBase = 'USD' AND c_in.date = with_normalized_currency.invoiceDate
           LEFT JOIN currency_exchange_rate c_out ON c_out.currencyTo = with_normalized_currency.normalizedCurrency AND c_out.currencyBase = 'USD' AND c_out.date = with_normalized_currency.invoiceDate
  WHERE with_normalized_currency.originalCurrency != with_normalized_currency.normalizedCurrency
)
, converted_invoice_normalize_pass_through AS (

    customerNumberHash,

    invoiceId,
    invoiceDate,

    customerEmail,
    amount,
    normalizedCurrency,
    originalAmount AS originalAmountNormalized
  FROM with_normalized_currency
  WHERE with_normalized_currency.originalCurrency = with_normalized_currency.normalizedCurrency
)
, converted_invoice_normalized AS (

, converted_invoice_target_currency_handler AS (

    converted_invoice_normalized.projectId,

    converted_invoice_normalized.customerNumberLC,

    converted_invoice_normalized.customerEmail,
    converted_invoice_normalized.amount,
    converted_invoice_normalized.normalizedCurrency,
    converted_invoice_normalized.originalAmountNormalized,
    CAST(CASE 
                 WHEN converted_invoice_normalized.originalCurrency = 'USD' THEN converted_invoice_normalized.originalAmount
                 ELSE ROUND((converted_invoice_normalized.amount * COALESCE(c_target.rate, 1)))
              END AS INT) AS converted_amount
  FROM converted_invoice_normalized
  LEFT JOIN currency_exchange_rate c_target ON c_target.currencyTo = 'USD'
           AND c_target.currencyBase = 'USD' AND c_target.date = converted_invoice_normalized.invoiceDate
  WHERE 'USD'!='USD'
)
, converted_invoice_target_currency_pass_through AS (

, rollforward_pre_records AS (
SELECT
  with_month_diff.projectId,
  with_month_diff.divisionId,
  with_month_diff.customerNumber,
  with_month_diff.customerNumberLC,
  with_month_diff.customerName,
  with_month_diff.customerEmail,
  with_month_diff.customerNumberHash,
  with_month_diff.parentCustomerNumber,
  with_month_diff.parentCustomerName,
  DATE_ADD(with_month_diff.startDate, INTERVAL with_month_diff.monthDiff MONTH) AS billDate,
  with_month_diff.invoiceDate,
  with_month_diff.invoiceId,
  with_month_diff.amount,
  with_month_diff.originalAmount,
  with_month_diff.nonRecurringAmount,
  with_month_diff.originalNonRecurringAmount,
  with_month_diff.originalCurrency AS currency,
  with_month_diff.integrationConfigId,
  IF(with_month_diff.originalAmount = 0,0,with_month_diff.amount/CAST(with_month_diff.originalAmount AS FLOAT)) AS exchangeRate,
  with_month_diff.internalPlanId AS planId,
  IF(with_month_diff.lineItemType IN ('adjustment-recurring', 'adjustment-nonrecurring'), 1, 0) AS hasAdjustment,
  ROW_NUMBER() OVER (PARTITION BY with_month_diff.projectId, with_month_diff.divisionId, with_month_diff.customerNumberLC ORDER BY with_month_diff.invoiceDate, with_month_diff.invoiceId, with_month_diff.amount) AS groupOrderNumber,
  ROW_NUMBER() OVER (PARTITION BY with_month_diff.projectId, with_month_diff.divisionId, with_month_diff.customerNumberLC ORDER BY with_month_diff.invoiceDate, with_month_diff.amount, with_month_diff.internalPlanId) AS groupOrderNumber2
FROM with_month_diff
)
, customer_monthly_totals AS (
SELECT
  projectId,
  divisionId,
  LOWER(customerNumber) AS customerNumberLC,
  month,
  externalPlanId,
  amount
FROM report_customer_revenue_rollforward
WHERE projectId = @projectId
         AND revenueModel = 'wayroll-gross-mrr'
         AND divisionId IN (1000093)
         
         AND 1=0
)
, customer_agg_by_month AS (
  SELECT
    rollforward_pre_records.projectId,
    rollforward_pre_records.divisionId,
    MAX_BY(customerNumber, groupOrderNumber) AS customerNumber,
    rollforward_pre_records.customerNumberLC,
    MAX_BY(customerName, groupOrderNumber) AS customerName,
    MAX_BY(customerEmail, groupOrderNumber) AS customerEmail,
    ANY_VALUE(customerNumberHash) AS customerNumberHash,
    MAX_BY(parentCustomerNumber, groupOrderNumber) AS parentCustomerNumber,
    MAX_BY(parentCustomerName, groupOrderNumber) AS parentCustomerName,
    CAST(MAX(billDate) AS DATE) AS billDate,
    CAST(MIN(billDate) AS DATE) AS minDate,
    CAST(DATE_ADD(billDate, INTERVAL -DAY(billDate) + 1 DAY) AS DATE) AS month,
    CAST(MAX(invoiceDate) AS DATE) AS invoiceDate,
    SUM(amount) AS amount,
    SUM(originalAmount) AS originalAmount,
    SUM(nonRecurringAmount) AS nonRecurringAmount,
    SUM(originalNonRecurringAmount) AS originalNonRecurringAmount,
    ANY_VALUE(currency) AS currency,
    MAX_BY(integrationConfigId,groupOrderNumber) AS integrationConfigId,
    IF(SUM(ABS(amount))=0,0,SUM(exchangeRate * ABS(amount))/SUM(ABS(amount))) AS exchangeRate,
    MAX_BY(planId, IF(amount > 0, groupOrderNumber2 + 1000000000, groupOrderNumber2)) AS planId,
    MAX(hasAdjustment) AS hasAdjustment
  FROM rollforward_pre_records
  GROUP BY projectId, divisionId, customerNumberLC, month
)

    customerNumberHash,
    parentCustomerNumber,
    parentCustomerName,
    billDate AS date,
    minDate,
    month,
    amount,
    originalAmount,
    nonRecurringAmount,
    originalNonRecurringAmount,
    currency,
    integrationConfigId,
    exchangeRate,
    planId,
    hasAdjustment
  FROM customer_agg_by_month
)
, rollforward_customer_records_for_union AS (
